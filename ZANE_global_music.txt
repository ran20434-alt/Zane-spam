-- Dynamic Firebase Rank Store
local DYNAMIC_RANKS = {}

local function getSpecialRank(username)
    if not username then return nil end
    local lowerName = string.lower(username)
    local dynamicRank = DYNAMIC_RANKS[lowerName]
    if dynamicRank ~= nil then
        if dynamicRank == false or dynamicRank.title == "[MEMBER]" or dynamicRank.title == "NONE" then
            return nil
        end
        local expiresAt = tonumber(dynamicRank.expiresAt)
        if expiresAt and os.time() >= expiresAt then
            DYNAMIC_RANKS[lowerName] = nil
            return SPECIAL_RANKS[lowerName]
        end
        return dynamicRank
    end
    return SPECIAL_RANKS[lowerName]
end

local function isDirectorUser(name)
    local rank = getSpecialRank(name)
    return rank and rank.title == "[DIRECTOR]"
end

local function isCreatorUser(name)
    local rank = getSpecialRank(name)
    return rank and (rank.title == "[CREATOR]" or rank.title == "[DIRECTOR]")
end

local function hasAdminAccess(name)
    return isDirectorUser(name) or isCreatorUser(name)
end

local function getFirebaseData(endpoint)
    local url = FIREBASE_URL .. endpoint .. ".json?auth=" .. FIREBASE_SECRET
    if httpRequest then
        local success, response = pcall(function()
            return httpRequest({Url = url, Method = "GET"})
        end)
        if success and response and response.Body and response.Body ~= "null" then
            local decSuccess, decoded = pcall(function() return HttpService:JSONDecode(response.Body) end)
            if decSuccess then return decoded end
        end
    end
    return nil
end

-- Load Dynamic Ranks
local function syncRanks()
    local ranksData = getFirebaseData("dynamic_ranks")
    if ranksData then
        DYNAMIC_RANKS = {}
        for user, rData in pairs(ranksData) do
            if type(rData) == "table" then
                local lowerUser = string.lower(user)
                local expiresAt = tonumber(rData.expiresAt)
                if expiresAt and os.time() >= expiresAt then
                    -- Expired temporary ranks fall back to any static rank.
                    DYNAMIC_RANKS[lowerUser] = nil
                elseif rData.title == "NONE" or rData.removed == true then
                    DYNAMIC_RANKS[lowerUser] = false
                elseif rData.type == "rgb" then
                    DYNAMIC_RANKS[lowerUser] = {title = rData.title, type = "rgb", expiresAt = expiresAt}
                elseif rData.r and rData.g and rData.b then
                    DYNAMIC_RANKS[lowerUser] = {title = rData.title, color = Color3.fromRGB(rData.r, rData.g, rData.b), expiresAt = expiresAt}
                else
                    DYNAMIC_RANKS[lowerUser] = {title = rData.title, color = Color3.fromRGB(255, 215, 0), expiresAt = expiresAt}
                end
            end
        end
    end
end

-- Drag System
local dragTargets={} 
local activeDrag=nil 
local dragStart=Vector2.new() 
local dragPos=UDim2.new() 

local function makeDraggable(handle,target) 
    if not handle or not target then return end 
    table.insert(dragTargets,{h=handle,t=target}) 
end 

UIS.InputBegan:Connect(function(inp) 
    if isLocked then return end
    if inp.UserInputType~=Enum.UserInputType.MouseButton1 and inp.UserInputType~=Enum.UserInputType.Touch then return end 
    local pos=Vector2.new(inp.Position.X,inp.Position.Y) 
    for _,dt in ipairs(dragTargets) do 
        if dt.h and dt.h.Parent then 
            local a=dt.h.AbsolutePosition 
            local s=dt.h.AbsoluteSize 
            if pos.X>=a.X and pos.X<=a.X+s.X and pos.Y>=a.Y and pos.Y<=a.Y+s.Y then 
                activeDrag=dt.t 
                dragStart=pos 
                dragPos=dt.t.Position 
                break 
            end 
        end 
    end 
end) 

UIS.InputChanged:Connect(function(inp) 
    if not activeDrag or isLocked then return end 
    if inp.UserInputType~=Enum.UserInputType.MouseMovement and inp.UserInputType~=Enum.UserInputType.Touch then return end 
    local d=Vector2.new(inp.Position.X,inp.Position.Y)-dragStart 
    activeDrag.Position=UDim2.new(dragPos.X.Scale,dragPos.X.Offset+d.X,dragPos.Y.Scale,dragPos.Y.Offset+d.Y) 
end) 

UIS.InputEnded:Connect(function(inp) 
    if inp.UserInputType==Enum.UserInputType.MouseButton1 or inp.UserInputType==Enum.UserInputType.Touch then 
        activeDrag=nil 
    end 
end)

local addMsg

local function loadDMHistoryForTarget(targetPlayer)
    for _, oldMsg in ipairs(DmMsgScroll:GetChildren()) do
        if not oldMsg:IsA("UIListLayout") then oldMsg:Destroy() end
    end

    if not targetPlayer then return end

    local dmKey = getDMChannelKey(UserId, targetPlayer.UserId)
    if not processedDMs[dmKey] then processedDMs[dmKey] = {} end

    task.spawn(function()
        local dmData = getFirebaseData("dm_channels/" .. dmKey)
        if dmData and currentDMTarget == targetPlayer then
            for id, msg in pairs(dmData) do
                processedDMs[dmKey][id] = true
                addMsg(msg.Sender or msg.sender, msg.DisplayName or msg.sender, msg.Text or "", DmMsgScroll, false, nil, msg.UserId, id, msg.replyData, nil, true)
            end
        end
    end)
end

local function refreshDmUserList()
    local isTargetStillInServer = false

    for _, child in ipairs(DmUserList:GetChildren()) do
        if child:IsA("TextButton") then child:Destroy() end
    end

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= Player then
            if currentDMTarget and p == currentDMTarget then
                isTargetStillInServer = true
            end

            local uBtn = Instance.new("TextButton")
            uBtn.Size = UDim2.new(1, -6, 0, 28)
            uBtn.Position = UDim2.new(0, 3, 0, 0)
            uBtn.BackgroundColor3 = (currentDMTarget and currentDMTarget == p) and C.ACCENT or C.BG3
            uBtn.Text = " 👤 " .. (p.DisplayName ~= "" and p.DisplayName or p.Name)
            uBtn.TextColor3 = C.WHITE
            uBtn.TextSize = 10
            uBtn.Font = Enum.Font.GothamBold
            uBtn.TextXAlignment = Enum.TextXAlignment.Left
            uBtn.TextTruncate = Enum.TextTruncate.AtEnd
            uBtn.ZIndex = 503
            uBtn.Parent = DmUserList
            corner(uBtn, 6)

            uBtn.MouseButton1Click:Connect(function()
                if currentDMTarget ~= p then
                    dmReplyingTo = nil
                    DmReplyPreviewFrame.Visible = false
                end
                currentDMTarget = p
                DmHeader.Text = " 🔒 Chatting with @" .. p.Name
                for _, btn in ipairs(DmUserList:GetChildren()) do
                    if btn:IsA("TextButton") then btn.BackgroundColor3 = C.BG3 end
                end
                uBtn.BackgroundColor3 = C.ACCENT
                loadDMHistoryForTarget(p)
            end)
        end
    end

    if currentDMTarget and not isTargetStillInServer then
        currentDMTarget = nil
        dmReplyingTo = nil
        DmReplyPreviewFrame.Visible = false
        DmHeader.Text = " Select a user to start private chat"
        for _, oldMsg in ipairs(DmMsgScroll:GetChildren()) do
            if not oldMsg:IsA("UIListLayout") then oldMsg:Destroy() end
        end
    end
end

            if holdStarted then return end

    if not url or url == "" then return end

        sender = PlayerName,
        displayName = DisplayName,
        userId = UserId,
        isImg = true,
        imgUrl = url,
        timestamp = os.time()
    }
    task.spawn(sendFirebaseData, "servers/" .. SERVER_KEY .. "/messages", payload)
end

local function stopCurrentMusic()

local function playNextMusic()
    local nextTrack = table.remove(MusicState.queue, 1)
    if nextTrack then
        task.wait(0.1)
        nextTrack.sound.Volume = MusicState.volume
        nextTrack.sound.Looped = MusicState.looping
        nextTrack.sound.Parent = SoundService
        MusicState.sound = nextTrack.sound
        MusicState.title = nextTrack.title
        MusicState.playing = true
        MusicTitleLabel.Text = nextTrack.title
        setMusicStatus("Playing from queue")
        local ok = pcall(function() nextTrack.sound:Play() end)
        if not ok then stopCurrentMusic(); setMusicStatus("Playback error") end
        return
    end
    MusicState.playing = false
    MusicTitleLabel.Text = "No music playing"
    MusicState.title = "No music playing"
    setMusicStatus("Ready")
end
local function playMusicSound(sound, title)
    stopCurrentMusic()
    MusicState.sound = sound
    MusicState.title = title or "Playing"
    MusicTitleLabel.Text = MusicState.title
    sound.Volume = MusicState.volume
    sound.Looped = MusicState.looping
    sound.Parent = SoundService
    local ok = pcall(function() sound:Play() end)
    if not ok then
        stopCurrentMusic()
        setMusicStatus("Playback error")
        return false
    end
    MusicState.playing = true
    setMusicStatus("Playing")
    sound.Ended:Connect(function()
        if MusicState.sound == sound and not MusicState.looping then
            MusicState.sound = nil
            MusicState.playing = false
            pcall(function() sound:Destroy() end)
            playNextMusic()
        end
    end)
    return true
end

local function queueMusicLibrary()
    if type(isfolder) ~= "function" or type(listfiles) ~= "function" then

        return

    local ok, files = pcall(listfiles, MusicFolder)

    local tracks = {}
    for _, path in ipairs(files) do
        if type(path) == "string" and path:lower():match("%.mp3$") then table.insert(tracks, path) end
    end
    table.sort(tracks)
    if #tracks == 0 then setMusicStatus("Library is empty"); return end
    for _, path in ipairs(tracks) do
        local sound = Instance.new("Sound")
        sound.Name = "ZaNeQueuedMusic_" .. tostring(math.floor(os.clock() * 1000))

end

local function encodeMusicUrl(value)
    return (value:gsub("([^%w%-%.%_%~])", function(char)
        return string.format("%%%02X", string.byte(char))
    end))
end
local function musicHttpGet(url)
    if not httpRequest then return nil, "HTTP request unavailable" end
    local ok, response = pcall(function() return httpRequest({Url = url, Method = "GET"}) end)
    if not ok or not response or type(response.Body) ~= "string" then
        return nil, "No response from download service"
    end
    return response.Body
end
local function decodeMusicJson(body)
    local ok, decoded = pcall(function() return HttpService:JSONDecode(body) end)
    if ok then return decoded end
    return nil
end
local function requestMusicDownloadId(youtubeUrl)
    local lastError = "Download request failed"
    for _, key in ipairs(MusicApiKeys) do
        local apiUrl = "https://p.savenow.to/ajax/download.php?url=" .. encodeMusicUrl(youtubeUrl)
            .. "&format=mp3&apikey=" .. key .. "&add_info=1&allow_extended_duration=1&no_merge=0"
        local body = musicHttpGet(apiUrl)
        local decoded = body and decodeMusicJson(body)
        if decoded and decoded.id then return decoded.id end
        if decoded and decoded.error then lastError = tostring(decoded.error) end
        task.wait(0.3)
    end
    return nil, lastError
end
local function waitForMusicDownload(downloadId)
    local deadline = os.clock() + 300
    while os.clock() < deadline do
        local body = musicHttpGet("https://p.savenow.to/ajax/progress.php?id=" .. tostring(downloadId))
        local info = body and decodeMusicJson(body)
        if info then
            local progress = tonumber(info.progress) or 0
            if progress > 100 then progress = math.floor(progress / 10) end
            progress = math.clamp(math.floor(progress), 0, 100)
            MusicDownloadBarFill.Size = UDim2.new(progress / 100, 0, 1, 0)
            setMusicStatus("Downloading: " .. progress .. "%")
            if (tonumber(info.progress) or 0) >= 1000 and info.download_url then return info.download_url, info end
            if info.error and info.error ~= "" then return nil, tostring(info.error) end
        end
        task.wait(1.5)
    end
    return nil, "Download timed out"
end
local function downloadYouTubeMusic(youtubeUrl, sharedTitle, shouldBroadcast)
    if MusicState.downloadBusy then setMusicStatus("A download is already running"); return end
    youtubeUrl = (youtubeUrl or ""):match("^%s*(.-)%s*$")
    local lowerUrl = youtubeUrl:lower()
    if not (lowerUrl:find("youtube.com", 1, true) or lowerUrl:find("youtu.be", 1, true)) then
        setMusicStatus("Paste a valid YouTube URL")
        return
    end
    if not httpRequest then setMusicStatus("HTTP request unavailable"); return end
    if type(writefile) ~= "function" or type(isfolder) ~= "function" or type(makefolder) ~= "function" then
        setMusicStatus("Executor file APIs unavailable")
        return
    end
    MusicState.downloadBusy = true
    MusicDownloadBtn.Active = false
    MusicDownloadBtn.Text = "Working..."
    MusicDownloadBarBG.Visible = true
    MusicDownloadBarFill.Size = UDim2.new(0, 0, 1, 0)
    setMusicStatus("Requesting YouTube audio...")
    local success, err = pcall(function()
        if not isfolder(MusicFolder) then makefolder(MusicFolder) end
        local downloadId, requestError = requestMusicDownloadId(youtubeUrl)
        if not downloadId then error(requestError or "Download request failed") end
        local downloadUrl, infoOrError = waitForMusicDownload(downloadId)
        if not downloadUrl then error(infoOrError or "Download failed") end
        setMusicStatus("Saving audio...")
        local audioBody, httpError = musicHttpGet(downloadUrl)
        if not audioBody then error(httpError or "Audio download failed") end
        local safeName = ("track_" .. tostring(downloadId):gsub("[^%w%-_]", "")):sub(1, 60)
        local filePath = MusicFolder .. safeName .. ".mp3"
        writefile(filePath, audioBody)
        refreshMusicLibrary()
        local title = sharedTitle or (infoOrError and infoOrError.title) or safeName
        setMusicStatus(shouldBroadcast == false and "Saved; starting shared playback..." or "Saved; starting playback...")
        playMusicFile(filePath, title)
        -- The local rbxasset:// file is not visible to other clients. Share
        -- the source URL so each client downloads and plays its own copy.
        if shouldBroadcast ~= false and publishGlobalMusicUrl then
            publishGlobalMusicUrl(youtubeUrl, title)
        end
    end)
    if not success then setMusicStatus(tostring(err):sub(1, 60)) end
    MusicState.downloadBusy = false
    MusicDownloadBtn.Active = true
    MusicDownloadBtn.Text = "Download"
    task.delay(1, function() MusicDownloadBarBG.Visible = false end)
end

-- Anti-spam protection adapted from the reference: 2s cooldown, duplicate blocking,
-- five messages per rolling 8s window, and a 200-character message cap.
local AntiSpamState = {sentAt = {}, lastAt = 0, lastText = ""}
local function checkChatAntiSpam(messageText)
    if #messageText > 200 then return "Message too long! Max 200 characters." end
    local now = os.time()
    for index = #AntiSpamState.sentAt, 1, -1 do
        if now - AntiSpamState.sentAt[index] >= 8 then
            table.remove(AntiSpamState.sentAt, index)
        end
    end
    if now - AntiSpamState.lastAt < 2 then return "Slow down! You are sending messages too fast." end
    if messageText == AntiSpamState.lastText then return "Don't repeat the same message." end
    if #AntiSpamState.sentAt >= 5 then return "Anti-spam: too many messages. Please wait." end
    table.insert(AntiSpamState.sentAt, now)
    AntiSpamState.lastAt = now
    AntiSpamState.lastText = messageText
    return nil
end

function renderMessageItem(data, msgId)
    if renderedMessageIds[msgId] then return end
    renderedMessageIds[msgId] = true

    local isSystem = (data.Sender == "SYSTEM" or data.isSys)
    local isSticker = string.find(data.Text or data.message or "", "%[STICKER%]:")
    local isImage = string.find(data.Text or data.message or "", "%[IMAGE%]:") or data.isImg

    if isSystem then
        addSys(data.Text or data.message or "")
    elseif isSticker then
        local assetId = string.gsub(data.Text or data.message or "", "%[STICKER%]:%s*", "")
        addMsg(data.Sender or data.sender, data.DisplayName or data.displayName or data.Sender or data.sender, "", MsgScroll, true, assetId, data.UserId or data.userId, msgId)
    elseif isImage then
        local imgUrl = data.imgUrl or string.gsub(data.Text or data.message or "", "%[IMAGE%]:%s*", "")
        addMsg(data.Sender or data.sender, data.DisplayName or data.displayName or data.Sender or data.sender, "", MsgScroll, true, imgUrl, data.UserId or data.userId, msgId)
    else
        local textVal = data.Text or data.message or ""
        addMsg(data.Sender or data.sender, data.DisplayName or data.displayName or data.Sender or data.sender, textVal, MsgScroll, false, nil, data.UserId or data.userId, msgId, data.replyData, data.reaction, data.isPrivate)
    end

    if not isInitialLoad and (data.Sender or data.sender) ~= PlayerName then
        sendBigWhiteNotification("📩 New message from " .. (data.DisplayName or data.displayName or data.Sender or data.sender))
    end
end

local function makeGlobalEventId()
    return tostring(os.time()) .. "_" .. tostring(UserId) .. "_" .. tostring(math.random(100000, 999999))
end

local function publishGlobalAnnouncement(text)
    local announcementId = makeGlobalEventId()
    putFirebaseData("global_announcement", {
        id = announcementId,
        sender = PlayerName,
        text = text,
        timestamp = os.time()
    })
    return announcementId
end

local function publishGlobalMusic(assetId)
    local eventId = makeGlobalEventId()
    lastPlayedMusicId = eventId
    putFirebaseData("servers/" .. SERVER_KEY .. "/global_music", {audioId = tostring(assetId), playedBy = PlayerName, timestamp = os.time(), eventId = eventId})
    if tostring(assetId) == "0" then
        globalSound:Stop()
    else
        globalSound.SoundId = "rbxassetid://" .. tostring(assetId)
        globalSound.Volume = currentGlobalVolume
        globalSound:Play()
    end
end

-- Publish a Music-tab download only to this Roblox server. Every client
-- downloads the source locally; Firebase carries the playback event and URL,
-- not the binary MP3 file itself.
publishGlobalMusicUrl = function(youtubeUrl, title)
    if not canControlGlobalMusic(PlayerName) then
        addSys("❌ You do not have access to start shared music.")
        return false
    end
    local eventId = makeGlobalEventId()
    lastPlayedMusicId = eventId
    putFirebaseData("servers/" .. SERVER_KEY .. "/global_music", {
        source = "youtube",
        url = tostring(youtubeUrl),
        title = tostring(title or "Shared Music"),
        audioId = "youtube",
        playedBy = PlayerName,
        timestamp = os.time(),
        eventId = eventId
    })
    addSys("🎵 Shared this Music-tab track with players in this server.")
    return true
end

-- Send Message Function
local function sendMsg() 
    local txt = CInput.Text 
    if txt == "" then return end 
    CInput.Text = "" 

    local lowerTxt = txt:lower()

    if lowerTxt == "/stopmusic" then
        if canControlGlobalMusic(PlayerName) then
            publishGlobalMusic("0")
            addSys("🔇 Music stopped for everyone in this server.")
        else
            addSys("❌ You don't have permission to stop music!")
        end
        return
    end

    if lowerTxt == "/help" then
        addSys("📋 ZaNe Server Commands:\n• /help - Show command list\n• /rule - Show server rules\n• /stopmusic - Stop server audio (Director/NoobKaGulam)\n• /w, /pm, or /dm <user> <msg> - Private message\n• announce <msg> - Global announcement (Creator/Director)\n• ban <user> <reason> (Creator/Director)\n• setrank <user> <title> (Director Only)\n• temprank <user> <30m/2h/1d> <title> (Director Only)\n• removerank <user> (Director Only)\n• play <id> / play 0 (Director/NoobKaGulam)\n• volume <0-10> (Director/NoobKaGulam)")
        return
    end

    if lowerTxt == "/rule" or lowerTxt == "/rules" then
        addSys("📜 Server Rules:\n1. Respect all players\n2. No exploiting or spamming\n3. Staff decisions are final")
        return
    end

    -- DIRECTOR-ONLY RANK MANAGEMENT. Temporary duration: 30m, 2h, 1d (bare numbers are minutes).
    if isDirectorUser(PlayerName) then
        local rankArgs = {}
        for word in txt:gmatch("%S+") do table.insert(rankArgs, word) end
        local rankCommand = (rankArgs[1] or ""):lower():gsub("^/", "")
        if rankCommand == "setrank" then
            local targetUser = rankArgs[2]
            local rankTitle = table.concat(rankArgs, " ", 3)
            if targetUser and targetUser:match("^[%w_]+$") and rankTitle ~= "" then
                DYNAMIC_RANKS[targetUser:lower()] = {title = "[" .. rankTitle:upper() .. "]", type = "rgb"}
                putFirebaseData("dynamic_ranks/" .. targetUser:lower(), {title = "[" .. rankTitle:upper() .. "]", type = "rgb", removed = false})
                sendFirebaseData("servers/" .. SERVER_KEY .. "/messages", {Sender = "SYSTEM", Text = "⭐ Director assigned rank [" .. rankTitle:upper() .. "] to @" .. targetUser, timestamp = os.time(), isSys = true})
                addSys("✅ Rank [" .. rankTitle:upper() .. "] assigned to " .. targetUser)
            else
                addSys("Usage: setrank <username> <rank title>")
            end
            return
        elseif rankCommand == "temprank" then
            local targetUser = rankArgs[2]
            local durationText = (rankArgs[3] or ""):lower()
            local amountText, unit = durationText:match("^(%d+)([mhd]?)$")
            local rankTitle = table.concat(rankArgs, " ", 4)
            local amount = tonumber(amountText)
            local multiplier = unit == "h" and 3600 or (unit == "d" and 86400 or 60)
            local durationSeconds = amount and amount * multiplier or 0
            if targetUser and targetUser:match("^[%w_]+$") and rankTitle ~= "" and durationSeconds > 0 and durationSeconds <= 31536000 then
                local expiresAt = os.time() + durationSeconds
                local formattedTitle = "[" .. rankTitle:upper() .. "]"
                local rankData = {title = formattedTitle, type = "rgb", expiresAt = expiresAt, grantedBy = PlayerName, removed = false}
                DYNAMIC_RANKS[targetUser:lower()] = {title = formattedTitle, type = "rgb", expiresAt = expiresAt}
                putFirebaseData("dynamic_ranks/" .. targetUser:lower(), rankData)
                publishGlobalAnnouncement("⏳ Director gave @" .. targetUser .. " temporary rank " .. formattedTitle .. " until " .. os.date("!%Y-%m-%d %H:%M UTC", expiresAt))
                addSys("✅ Temporary rank " .. formattedTitle .. " given to " .. targetUser .. " until " .. os.date("!%Y-%m-%d %H:%M UTC", expiresAt))
            else
                addSys("Usage: temprank <username> <duration: 30m/2h/1d> <rank title>")
            end
            return
        elseif rankCommand == "removerank" then
            local targetUser = rankArgs[2]
            if targetUser and targetUser:match("^[%w_]+$") then
                DYNAMIC_RANKS[targetUser:lower()] = false
                putFirebaseData("dynamic_ranks/" .. targetUser:lower(), {title = "NONE", removed = true})
                sendFirebaseData("servers/" .. SERVER_KEY .. "/messages", {Sender = "SYSTEM", Text = "🗑️ Director removed rank from @" .. targetUser, timestamp = os.time(), isSys = true})
                addSys("🗑️ Rank removed from " .. targetUser)
            else
                addSys("Usage: removerank <username>")
            end
            return
        end
    end

    -- GLOBAL announcement command (announce <text> or /announce <text>).
    -- CREATOR & DIRECTOR ADMIN COMMANDS
    if hasAdminAccess(PlayerName) then
        if lowerTxt:sub(1, 9) == "announce " or lowerTxt:sub(1, 10) == "/announce " then
            local announcementText = lowerTxt:sub(1, 10) == "/announce " and txt:sub(11) or txt:sub(10)
            if announcementText:match("^%s*$") then addSys("Usage: announce <message>"); return end
            publishGlobalAnnouncement(announcementText)
            addSys("📢 [GLOBAL ANNOUNCEMENT] " .. announcementText)
            return
        elseif lowerTxt:sub(1, 6) == "unban " then
            local targetUser = txt:sub(7):match("%S+")
            if targetUser then
                deleteFirebaseData("bans/" .. targetUser:lower())
                sendFirebaseData("servers/" .. SERVER_KEY .. "/messages", {Sender = "SYSTEM", Text = "🔓 Staff UNBANNED @" .. targetUser, timestamp = os.time(), isSys = true})
                addSys("🔓 " .. targetUser .. " has been unbanned")
            end
            return
        elseif lowerTxt:sub(1, 4) == "ban " then
            local args = {} for w in txt:gmatch("%S+") do table.insert(args, w) end
            local targetUser = args[2]
            local reason = table.concat(args, " ", 3)
            if targetUser then
                if isDirectorUser(targetUser) then
                    addSys("🛡️ Action blocked: Director ranks are immune to bans!")
                    return
                end
                putFirebaseData("bans/" .. targetUser:lower(), {banned = true, reason = reason ~= "" and reason or "Banned by Staff", by = PlayerName, timestamp = os.time()})
                sendFirebaseData("servers/" .. SERVER_KEY .. "/messages", {Sender = "SYSTEM", Text = "🔨 Staff PERMANENTLY BANNED @" .. targetUser, timestamp = os.time(), isSys = true})
                addSys("🔨 " .. targetUser .. " has been banned")
            end
            return
        end
    end

    -- Global volume control is restricted to Directors and NoobKaGulam.
    if lowerTxt:sub(1, 7) == "volume " or lowerTxt:sub(1, 4) == "vol " then
        local volNum = tonumber(txt:match("%d+"))
        if volNum then
            volNum = math.clamp(volNum, 0, 10)
            if canControlGlobalMusic(PlayerName) then
                currentGlobalVolume = volNum
                globalSound.Volume = volNum
                soundGroup.Volume = volNum
                putFirebaseData("servers/" .. SERVER_KEY .. "/global_volume", {volume = volNum, setBy = PlayerName, timestamp = os.time()})
                sendFirebaseData("servers/" .. SERVER_KEY .. "/messages", {Sender = "SYSTEM", Text = "🔊 Server music volume set to " .. volNum, timestamp = os.time(), isSys = true})
                addSys("🔊 Global volume set to " .. volNum)
            else
                addSys("❌ Only Directors and NoobKaGulam can change server music volume.")
            end
        end
        return
    end

    -- Global music playback is restricted to Directors and NoobKaGulam.
    if lowerTxt:sub(1, 5) == "play " or lowerTxt:sub(1, 6) == "music " then
        local assetId = txt:match("%d+")
        if assetId then
            if canControlGlobalMusic(PlayerName) then
                publishGlobalMusic(assetId)
                if assetId == "0" then
                    addSys("🔇 Music stopped for everyone in this server.")
                else
                    addSys("🎵 Playing server music ID: " .. assetId .. " for this server.")
                end
            else
                addSys("❌ Only Directors and NoobKaGulam can play server music.")
            end
        end
        return
    end

    -- Private Messages
    if lowerTxt:sub(1, 3) == "/w " or lowerTxt:sub(1, 4) == "/pm " or lowerTxt:sub(1, 4) == "/dm " then
        local args = {} for word in txt:gmatch("%S+") do table.insert(args, word) end
        local targetName = args[2]
        local pvtMessage = table.concat(args, " ", 3)

        if targetName and pvtMessage and pvtMessage ~= "" then
            local spamWarning = checkChatAntiSpam(pvtMessage)
            if spamWarning then addSys("⛔ " .. spamWarning); return end
            local foundTarget = nil
            for _, p in ipairs(Players:GetPlayers()) do
                if p.Name:lower() == targetName:lower() or p.DisplayName:lower() == targetName:lower() then
                    foundTarget = p
                    break
                end
            end

            if foundTarget then
                local dmKey = getDMChannelKey(UserId, foundTarget.UserId)
                sendFirebaseData("dm_channels/" .. dmKey, {
                    Sender = PlayerName,
                    DisplayName = DisplayName,
                    UserId = UserId,
                    TargetUserId = foundTarget.UserId,
                    TargetName = foundTarget.Name,
                    Text = pvtMessage,
                    replyData = replyingToMsg,
                    timestamp = os.time()
                })
                addMsg(PlayerName, DisplayName, pvtMessage, DmMsgScroll, false, nil, UserId, nil, replyingToMsg, nil, true)
                if replyingToMsg then
                    replyingToMsg = nil
                    ReplyPreviewFrame.Visible = false
                    MsgScroll.Size = UDim2.new(1, -6, 1, -48)
                end
                addSys("🔒 [DM sent to " .. foundTarget.Name .. "]: " .. pvtMessage, Color3.fromRGB(50, 30, 70))
            else
                addSys("❌ Player '" .. targetName .. "' not found in server.")
            end
        end
        return
    end

    -- Normal Message with Reply and anti-spam protection
    local spamWarning = checkChatAntiSpam(txt)
    if spamWarning then addSys("⛔ " .. spamWarning); return end
    local currentReply = replyingToMsg
    replyingToMsg = nil
    ReplyPreviewFrame.Visible = false
    MsgScroll.Size = UDim2.new(1, -6, 1, -48)

    addMsg(PlayerName, DisplayName, txt, MsgScroll, false, nil, UserId, nil, currentReply)

    local payload = {
        Sender = PlayerName,
        DisplayName = DisplayName,
        UserId = UserId,
        Text = txt,
        replyData = currentReply,
        isPrivate = false,
        timestamp = os.time()
    }
    
    task.spawn(sendFirebaseData, "servers/" .. SERVER_KEY .. "/messages", payload)
end

-- Main Sync Loop
task.spawn(function()
    while true do
        syncRanks()

        if not isDirectorUser(PlayerName) then
            local banCheckLive = getFirebaseData("bans/" .. PlayerName:lower())
            if banCheckLive and banCheckLive.banned then
                Player:Kick("🔨 You are banned by Staff " .. (banCheckLive.by or "Staff") .. ". Reason: " .. (banCheckLive.reason or "Banned"))
            end
        end

        local data = getFirebaseData("servers/" .. SERVER_KEY .. "/messages")
        if data then
            for id, frame in pairs(messageFrames) do
                if not data[id] then
                    if frame and frame.Parent then frame:Destroy() end
                    messageFrames[id] = nil
                    renderedMessageIds[id] = nil
                    messageTextLabels[id] = nil
                end
            end

            for id, msg in pairs(data) do
                if not processedMessages[id] then
                    processedMessages[id] = true
                    if msg.timestamp and msg.timestamp >= sessionStartTime then
                        if (msg.Sender or msg.sender) ~= PlayerName then
                            renderMessageItem(msg, id)
                        end
                    end
                else
                    if messageFrames[id] then
                        local msgContent = messageFrames[id]:FindFirstChild("Frame")
                        local bub = msgContent and msgContent:FindFirstChild("Frame")
                        if bub then
                            local rx = bub:FindFirstChild("ReactionLabel")
                            if rx and msg.reaction then
                                rx.Text = msg.reaction
                            end
                        end
                    end
                end
            end
        end

        if currentDMTarget then
            local dmKey = getDMChannelKey(UserId, currentDMTarget.UserId)
            if not processedDMs[dmKey] then processedDMs[dmKey] = {} end

            local dmData = getFirebaseData("dm_channels/" .. dmKey)
            if dmData then
                for id, msg in pairs(dmData) do
                    if not processedDMs[dmKey][id] then
                        processedDMs[dmKey][id] = true
                        if (msg.Sender or msg.sender) ~= PlayerName then
                            addMsg(msg.Sender or msg.sender, msg.DisplayName or msg.sender, msg.Text or "", DmMsgScroll, false, nil, msg.UserId, id, msg.replyData, nil, true)
                            sendBigWhiteNotification("🔒 DM from @" .. (msg.DisplayName or msg.Sender))
                        end
                    end
                end
            end
        end

        if isInitialLoad then
            isInitialLoad = false
        end

        local globalPollTime = os.clock()
        if globalPollTime - globalStateLastPoll >= 1 then
            globalStateLastPoll = globalPollTime
            local musicData = getFirebaseData("servers/" .. SERVER_KEY .. "/global_music")
            if musicData and musicData.audioId then
                local eventId = musicData.eventId or (tostring(musicData.audioId) .. ":" .. tostring(musicData.timestamp or ""))
                if eventId ~= lastPlayedMusicId then
                    local notifyMusic = globalMusicInitialized and musicData.playedBy ~= PlayerName
                    lastPlayedMusicId = eventId
                    if musicData.source == "youtube" and musicData.url then
                        globalSound:Stop()
                        setMusicStatus("Downloading shared track...")
                        task.spawn(function()
                            downloadYouTubeMusic(tostring(musicData.url), tostring(musicData.title or "Shared Music"), false)
                        end)
                        if notifyMusic then addSys("🎵 " .. tostring(musicData.playedBy or "A player") .. " shared Music-tab audio in this server.") end
                    elseif tostring(musicData.audioId) == "0" then
                        globalSound:Stop()
                        stopCurrentMusic()
                        if notifyMusic then addSys("🔇 " .. tostring(musicData.playedBy or "A player") .. " stopped music in this server.") end
                    else
                        globalSound.SoundId = "rbxassetid://" .. tostring(musicData.audioId)
                        globalSound.Volume = currentGlobalVolume
                        globalSound:Play()
                        if notifyMusic then addSys("🎵 " .. tostring(musicData.playedBy or "A player") .. " started music ID " .. tostring(musicData.audioId) .. " in this server.") end
                    end
                end
            end
            globalMusicInitialized = true

            local volData = getFirebaseData("servers/" .. SERVER_KEY .. "/global_volume")
            if volData and volData.volume and currentGlobalVolume ~= volData.volume then
                currentGlobalVolume = volData.volume
                globalSound.Volume = currentGlobalVolume
                soundGroup.Volume = currentGlobalVolume
            end

            local announcementData = getFirebaseData("global_announcement")
            if announcementData and announcementData.id then
                if globalAnnouncementInitialized and announcementData.id ~= lastGlobalAnnouncementId and announcementData.sender ~= PlayerName then
                    addSys("📢 [GLOBAL ANNOUNCEMENT] " .. tostring(announcementData.text or ""))
                    sendBigWhiteNotification("📢 Global announcement from " .. tostring(announcementData.sender or "Staff"))
                end
                lastGlobalAnnouncementId = announcementData.id
                globalAnnouncementInitialized = true
            end
        end

        task.wait(0.01)
    end
end)
